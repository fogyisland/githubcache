# M19 — Providers (JSON-Driven Ingestion Sources) Implementation Plan

> **For agentic workers:** REQUIRED SUB-KILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a new admin concept "ingestion provider" so operators can declaratively configure JSON-driven data sources whose contents are parsed into GitHub `owner/name` pairs and queued for refresh into the cache — end-to-end testable even when the GitHub token pool is empty.

**Architecture:** New DB table `ingestion_providers` (slug, kind='json', JSONPath-based config, enabled). Admin UI at `/admin/providers` for CRUD; ingestion page gets a new card "Run via provider" that calls `POST /api/admin/providers/[id]/preview` (parse + dedupe + report) and `POST /api/admin/providers/[id]/run` (enqueue refresh jobs). Parsing/extraction/dedupe live in `src/lib/ingestion/providers/`.

**Tech Stack:** Next.js 14 app router (RSC) + Prisma 5 (MySQL 5.7.44) + Zod 4 + next-intl 4 + Vitest 4 + existing `refresh_jobs` scheduler.

**Spec:** `docs/superpowers/specs/2026-08-31-m19-providers.md`

export class JsonPathError extends Error {
  constructor(path: string, message: string) {
    super(`Invalid JSONPath "${path}": ${message}`);
    this.name = 'JsonPathError';
  }
}

/**
 * Minimal JSONPath evaluator supporting the subset needed for M19 providers.
 * Recognized forms:
 *   $.a.b.c          — property access
 *   $.a[0]           — index access on the *current* segment
 *   $.a[*].b         — wildcard index over an array, then property
 *
 * Does NOT support: $..desc, filter expressions [?(...)], script
 * expressions, multi-index [0,1], or slice [0:3].
 */
export function evaluateJsonPath(input: unknown, path: string): unknown {
  if (!path.startsWith('$')) {
    throw new JsonPathError(path, 'must start with "$"');
  }
  // Strip leading "$" and walk the remainder in segments.
  const remainder = path.slice(1);
  if (remainder === '') return input;
  // Tokenize: each segment is either `.word` or `[N]` or `[*]`.
  const tokenRe = /\.([A-Za-z_][A-Za-z0-9_]*)|\[(\d+|\*)\]/g;
  const tokens: Array<{ kind: 'key'; value: string } | { kind: 'index' | 'wildcard'; value: number | '*' }> = [];
  let m: RegExpExecArray | null;
  while ((m = tokenRe.exec(remainder)) !== null) {
    if (m[1] !== undefined) tokens.push({ kind: 'key', value: m[1] });
    else if (m[2] === '*') tokens.push({ kind: 'wildcard', value: '*' });
    else tokens.push({ kind: 'index', value: Number(m[2]) });
  }
  // The remainder after stripping tokens must be empty.
  const cleaned = remainder.replace(tokenRe, '');
  if (cleaned !== '') {
    throw new JsonPathError(path, 'unrecognized segment near "' + cleaned + '"');
  }
  let cur: unknown = input;
  for (const tok of tokens) {
    if (tok.kind === 'key') {
      if (cur === null || typeof cur !== 'object') {
        throw new JsonPathError(path, `cannot read property "${tok.value}" on ${typeof cur}`);
      }
      cur = (cur as Record<string, unknown>)[tok.value];
    } else if (tok.kind === 'index') {
      if (!Array.isArray(cur)) {
        throw new JsonPathError(path, `cannot index ${typeof cur} with [${tok.value}]`);
      }
      cur = cur[tok.value as number];
    } else {
      // wildcard: return array of values for next key step (handled by caller)
      // For a wildcard token, return the array; the NEXT key token will map over it.
      if (!Array.isArray(cur)) {
        throw new JsonPathError(path, 'cannot wildcard over non-array');
      }
      // Mark by returning a wrapper; downstream key applies element-wise.
      // Simpler approach: pre-resolve by consuming the following key token inline.
      throw new JsonPathError(path, 'wildcard [*] must be immediately followed by a key access in this implementation');
  }
  return cur;
}

export function evaluateJsonPath(input: unknown, path: string): unknown {
  if (!path.startsWith('$')) throw new JsonPathError(path, 'must start with "$"');
  const remainder = path.slice(1);
  let cur: unknown = input;
  let i = 0;
  while (i < remainder.length) {
    if (remainder[i] === '.') {
      const m = /^([A-Za-z_][A-Za-z0-9_]*)/.exec(remainder.slice(i + 1));
      if (!m) throw new JsonPathError(path, 'expected property name after "."');
      if (cur === null || typeof cur !== 'object') {
        throw new JsonPathError(path, `cannot read "${m[1]}" on ${cur === null ? 'null' : typeof cur}`);
      }
      cur = (cur as Record<string, unknown>)[m[1]];
      i += 1 + m[1].length;
    } else if (remainder[i] === '[') {
      const close = remainder.indexOf(']', i);
      if (close === -1) throw new JsonPathError(path, 'unmatched "["');
      const inside = remainder.slice(i + 1, close);
      if (inside === '*') {
        if (!Array.isArray(cur)) throw new JsonPathError(path, 'cannot [*] over non-array');
        // Consume an immediately-following .key if present.
        let nextKey: string | null = null;
        if (remainder[close + 1] === '.') {
          const m = /^([A-Za-z_][A-Za-z0-9_]*)/.exec(remainder.slice(close + 2));
          if (m) { nextKey = m[1]; i = close + 2 + m[1].length; }
          else { i = close + 1; }
        } else {
          i = close + 1;
        }
        const arr = cur as unknown[];
        if (nextKey === null) {
          cur = arr;
        } else {
          cur = arr.map((el) => (el !== null && typeof el === 'object' ? (el as Record<string, unknown>)[nextKey!] : undefined));
        }
      } else if (/^\d+$/.test(inside)) {
        if (!Array.isArray(cur)) throw new JsonPathError(path, `cannot [${inside}] over non-array`);
        cur = cur[Number(inside)];
        i = close + 1;
      } else {
        throw new JsonPathError(path, `unsupported index expression "[${inside}]"`);
      }
    } else {
      throw new JsonPathError(path, `unexpected character "${remainder[i]}"`);
    }
  }
  return cur;
}

import { promises as fs } from 'node:fs';
import path from 'node:path';
import { evaluateJsonPath, JsonPathError } from './jsonpath';
import type { ProviderConfig } from './schema';

export class ProviderSourceError extends Error {
  constructor(message: string, public readonly code: 'FILE_NOT_FOUND' | 'FILE_OUT_OF_ROOTS' | 'HTTP_ERROR' | 'JSON_PARSE' | 'BAD_ITEMSPATH' | 'BAD_URLFIELD') {
    super(message);
    this.name = 'ProviderSourceError';
  }
}

function resolveFileRoots(): string[] {
  const raw = process.env.PROVIDER_FILE_ROOTS ?? '';
  return raw.split(',').map((s) => s.trim()).filter(Boolean).map((p) => path.resolve(p));
}

function isUnder(child: string, parents: readonly string[]): boolean {
  const abs = path.resolve(child);
  return parents.some((p) => abs === p || abs.startsWith(p + path.sep));
}

async function readJsonFile(p: string, fileRoots: readonly string[]): Promise<unknown> {
  const abs = path.resolve(p);
  if (!isUnder(abs, fileRoots)) {
    throw new ProviderSourceError(`path "${p}" is outside PROVIDER_FILE_ROOTS`, 'FILE_OUT_OF_ROOTS');
  }
  let raw: string;
  try { raw = await fs.readFile(abs, 'utf8'); }
  catch (e: unknown) {
    const code = (e as NodeJS.ErrnoException).code;
    if (code === 'ENOENT') throw new ProviderSourceError(`file not found: ${p}`, 'FILE_NOT_FOUND');
    throw new ProviderSourceError(`failed to read ${p}: ${(e as Error).message}`, 'FILE_NOT_FOUND');
  }
  try { return JSON.parse(raw); }
  catch (e: unknown) {
    throw new ProviderSourceError(`invalid JSON in ${p}: ${(e as Error).message}`, 'JSON_PARSE');
  }
}

async function fetchJson(url: string, headers: Record<string, string> | undefined, signal: AbortSignal | undefined): Promise<unknown> {
  const init: RequestInit = { method: 'GET', headers };
  if (signal) init.signal = signal;
  const res = await fetch(url, init);
  if (!res.ok) throw new ProviderSourceError(`HTTP ${res.status} fetching ${url}`, 'HTTP_ERROR');
  const text = await res.text();
  try { return JSON.parse(text); }
  catch (e: unknown) {
    throw new ProviderSourceError(`invalid JSON from ${url}: ${(e as Error).message}`, 'JSON_PARSE');
  }
}

export interface LoadSourceOptions {
  signal?: AbortSignal;
  fileRoots?: readonly string[];
}

export async function loadProviderSource(config: ProviderConfig, opts: LoadSourceOptions = {}): Promise<unknown[]> {
  const fileRoots = opts.fileRoots ?? resolveFileRoots();
  let raw: unknown;
  if (config.kind === 'file') {
    raw = await readJsonFile(config.path, fileRoots);
  } else {
    raw = await fetchJson(config.url, config.headers, opts.signal);
  }
  let items: unknown;
  try { items = evaluateJsonPath(raw, config.itemsPath); }
  catch (e: unknown) {
    if (e instanceof JsonPathError) {
      throw new ProviderSourceError(`itemsPath "${config.itemsPath}": ${e.message}`, 'BAD_ITEMSPATH');
    }
    throw e;
  }
  if (!Array.isArray(items)) {
    throw new ProviderSourceError(`itemsPath "${config.itemsPath}" did not yield an array`, 'BAD_ITEMSPATH');
  }
  return items;
}

  invalid: number;

  stale: number;
  new: number;

const SAMPLE_LIMIT = 20;

async function loadProviderOrThrow(slug: string) {
  const provider = await prisma.ingestionProvider.findUnique({ where: { slug } });
  if (!provider) {
    const e = new Error(`provider not found: ${slug}`);
    (e as Error & { code: string }).code = 'NOT_FOUND';
    throw e;
  }
  if (!provider.enabled) {
    const e = new Error(`provider disabled: ${slug}`);
    (e as Error & { code: string }).code = 'DISABLED';
    throw e;
  }
  return provider;
}

function classify(records: Array<{ owner: string; name: string; status: 'ok' | 'stale' | 'absent' }>): Pick<PreviewTotals, 'existing' | 'stale' | 'new'> {
  return {
    existing: records.filter((r) => r.status === 'ok').length,
    stale: records.filter((r) => r.status === 'stale').length,
    new: records.filter((r) => r.status === 'absent').length,
  };
}

export async function previewProvider(slug: string, opts: { limit?: number } = {}): Promise<PreviewResult> {
  const provider = await loadProviderOrThrow(slug);
  const config: ProviderConfig = parseProviderConfig(provider.configJson);
  const items = await loadProviderSource(config);
  const urls: string[] = [];
  for (const item of items) {
    let v: unknown;
    try { v = evaluateUrlField(item, config.urlField); }
    catch { continue; }
    if (typeof v === 'string' && v.length > 0) urls.push(v);
  }
  const pairs = urls.map(parseGitHubUrl).filter((p): p is { owner: string; name: string } => p !== null);
  const uniquePairs = dedupe(pairs);
  const limited = opts.limit ? uniquePairs.slice(0, opts.limit) : uniquePairs;
  const rows = await prisma.repository.findMany({
    where: { OR: limited.map((p) => ({ owner: p.owner, name: p.name })) },
    select: { owner: true, name: true, fetchStatus: true, lastFetchedAt: true },
  });
  const rowMap = new Map(rows.map((r) => [`${r.owner}/${r.name}`, r]));
  const classified = limited.map((p) => {
    const row = rowMap.get(`${p.owner}/${p.name}`);
    let s: 'ok' | 'stale' | 'absent';
    if (!row) s = 'absent';
    else if (row.fetchStatus !== 'ok' || row.lastFetchedAt === null) s = 'stale';
    else s = 'ok';
    return { ...p, status: s };
  });
  return {
    totals: {
      items: items.length,
      urls: urls.length,
      unique: uniquePairs.length,
      invalid: urls.length - pairs.length,
      ...classify(classified),
    },
    sample: classified.filter((c) => c.status !== 'ok').slice(0, SAMPLE_LIMIT).map(({ owner, name }) => ({ owner, name })),
  };
}

export async function runProvider(slug: string, opts: { limit?: number; dryRun?: boolean } = {}): Promise<RunResult> {
  const preview = await previewProvider(slug, { limit: opts.limit });
  if (opts.dryRun) {
    return { totals: preview.totals, jobCount: 0, dryRun: true };
  }
  // Re-load to get fresh rows + full set of to-be-enqueued (stale + new).
  const provider = await loadProviderOrThrow(slug);
  const config: ProviderConfig = parseProviderConfig(provider.configJson);
  const items = await loadProviderSource(config);
  const urls: string[] = [];
  for (const item of items) {
    let v: unknown;
    try { v = evaluateUrlField(item, config.urlField); }
    catch { continue; }
    if (typeof v === 'string' && v.length > 0) urls.push(v);
  }
  const pairs = urls.map(parseGitHubUrl).filter((p): p is { owner: string; name: string } => p !== null);
  const uniquePairs = dedupe(pairs);
  const limited = opts.limit ? uniquePairs.slice(0, opts.limit) : uniquePairs;
  const rows = await prisma.repository.findMany({
    where: { OR: limited.map((p) => ({ owner: p.owner, name: p.name })) },
    select: { id: true, owner: true, name: true },
  });
  const rowMap = new Map(rows.map((r) => [`${r.owner}/${r.name}`, r]));
  const toEnqueue: { owner: string; name: string }[] = [];
  for (const p of limited) {
    if (!rowMap.has(`${p.owner}/${p.name}`)) toEnqueue.push(p);
  }
  // Ensure repositories exist (upsert stub rows) then create refresh_jobs.
  let jobCount = 0;
  if (toEnqueue.length > 0) {
    await prisma.$transaction(async (tx) => {
      for (const p of toEnqueue) {
        const repo = await tx.repository.upsert({
          where: { owner_name: { owner: p.owner, name: p.name } },
          create: { owner: p.owner, name: p.name, node: {} as never, fetchStatus: 'ok' },
          update: {},
        });
        await tx.refreshJob.create({
          data: {
            repositoryId: repo.id,
            priority: 70,
            scheduledFor: new Date(),
          },
        });
        jobCount++;
      }
    });
  }
  // Also re-enqueue existing stale rows so they refresh too.
  const stalePairs = limited.filter((p) => rowMap.has(`${p.owner}/${p.name}`));
  if (stalePairs.length > 0) {
    await prisma.$transaction(async (tx) => {
      for (const p of stalePairs) {
        const repo = rowMap.get(`${p.owner}/${p.name}`)!;
        await tx.refreshJob.create({
          data: {
            repositoryId: repo.id,
            priority: 70,
            scheduledFor: new Date(),
          },
        });
        jobCount++;
      }
    });
  }
  return { totals: preview.totals, jobCount, dryRun: false };
}

function dedupe(pairs: { owner: string; name: string }[]): { owner: string; name: string }[] {
  const seen = new Set<string>();
  const out: { owner: string; name: string }[] = [];
  for (const p of pairs) {
    const key = `${p.owner}/${p.name}`;
    if (seen.has(key)) continue;
    seen.add(key);
    out.push(p);
  }
  return out;
}

// urlField semantics: supports 'field', 'field.subfield', 'field[N]', 'field[*]'.
function evaluateUrlField(item: unknown, urlField: string): unknown {
  // Accept JSONPath-style: itemsPath already extracted an array; here we apply urlField to each item.
  // We treat urlField as a JSONPath rooted at "$". Reuse evaluateJsonPath.
  const { evaluateJsonPath } = require('./jsonpath') as typeof import('./jsonpath');
  return evaluateJsonPath({ $: item }, '$' + urlField.replace(/^\$/, ''));
}
